create table if not exists public.service_assessments (
  id text primary key,
  created_at timestamptz not null default now(),
  assessee text,
  assessor text,
  assessed_at text,
  total numeric,
  rate numeric,
  payload jsonb
);

alter table public.service_assessments enable row level security;

drop policy if exists "assess_anon_insert" on public.service_assessments;
create policy "assess_anon_insert" on public.service_assessments
  for insert to anon with check (true);

drop policy if exists "assess_anon_select" on public.service_assessments;
create policy "assess_anon_select" on public.service_assessments
  for select to anon using (true);

drop policy if exists "assess_anon_update" on public.service_assessments;
create policy "assess_anon_update" on public.service_assessments
  for update to anon using (true) with check (true);
